import time
import colorama
from colorama import Fore, Back, Style
import random

colorama.init()

class Character:
    def __init__(self, health, regen, damage):
        self.health = health
        self.regen = regen
        self.damage = damage

player = Character(200, 10, 15)
warrior = Character(100, 15, 20)

# Game functions

def game_status():
    print("\n" + f"{Fore.YELLOW}="*40)
    print(f"{Fore.CYAN}You     HP: {player.health}/{200}")
    print(f"{Fore.RED}Warrior HP: {warrior.health}/{100}")
    print(f"{Fore.YELLOW}="*40)

# Player def

def player_attack():
    time.sleep(0.5)
    print(Fore.GREEN, Style.BRIGHT + f"Great! You atacked the warrior and caused {player.damage} damage! Warrior has {warrior.health - player.damage} health left!")
    warrior.health -= player.damage

def player_health():
    time.sleep(1)
    if player.health < 200:
        print(Fore.GREEN, Style.BRIGHT + f"Regen health, you have {player.health} health!")
        player.health += player.regen

# Warrior defs

def warrior_attack():
    time.sleep(0.5)
    print(Fore.RED, Style.BRIGHT + f"Ho no! Warrior atacked you and caused {warrior.damage} damage! You have {player.health - warrior.damage} health left!")
    player.health -= warrior.damage

def warrior_health():
    if warrior.health < 100:
        time.sleep(1)
        print(Fore.RED, Style.BRIGHT + f"Warrior regen his health, warrior has {warrior.health} health!")
        warrior.health += warrior.regen

    Player_name = input(Fore.YELLOW + Style.BRIGHT + "--> ").strip()

    if len(Player_name) == 0:
        print(Fore.RED + "Name cannot be empty!")
        continue

    if len(Player_name) > 20:
        print(Fore.RED + f"Name too long! ({len(Player_name)} characters — max 20)")
        continue

    break

time.sleep(1.5)

print(Fore.YELLOW, Style.BRIGHT + f"\rWelcome {Player_name}!", end="", flush=True)

while True:
    game_status()
    print(Fore.YELLOW, Style.BRIGHT + "\nWhat to do?   [0] Quit    [1] Attack    [2] Regen    ")
    
    try:
        choice = int(input("--> "))
    except ValueError:
        print(Fore.RED + "Invalid! Enter 1, 2 or 3.")
        continue
    
    if choice == 0:
        print("Goodbye!")
        break
    if choice not in (1, 2):
        print(Fore.RED + "Choose 1 or 2!")
        continue
    
    # Player acts
    if choice == 1:
        player_attack()
    else:
        player_health()
    
    # Warrior acts
    warrior_choice = random.randint(1, 2)
    if warrior_choice == 1:
        warrior_attack()
    else:
        warrior_health()
    
    # Win or lose?
    if player.health <= 0:
        player.health = 0
        print(Fore.LIGHTRED_EX + "You died! Game over.")
        game_status()
        break
    if warrior.health <= 0:
        warrior.health = 0
        print(Fore.LIGHTGREEN_EX + "Warrior died! Victory!")
        game_status()
        break
